# The built-in font, compiled into the binary so every game and tool has text
# without shipping a file: Atkinson Hyperlegible, SIL Open Font License 1.1
# (fonts/OFL.txt). Regenerated when the .ttf changes.
set(AIGE_FONT_TTF ${CMAKE_CURRENT_SOURCE_DIR}/fonts/AtkinsonHyperlegible-Regular.ttf)
set(AIGE_FONT_CPP ${CMAKE_CURRENT_BINARY_DIR}/BuiltinFontData.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${AIGE_FONT_TTF})
file(READ ${AIGE_FONT_TTF} AIGE_FONT_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," AIGE_FONT_BYTES "${AIGE_FONT_HEX}")
file(WRITE ${AIGE_FONT_CPP}.tmp
    "// Generated by engine/renderer2d/CMakeLists.txt from ${AIGE_FONT_TTF}\n"
    "#include <cstddef>\n#include <cstdint>\n"
    "namespace aige {\n"
    "extern const uint8_t kBuiltinFontData[] = {${AIGE_FONT_BYTES}};\n"
    "extern const size_t kBuiltinFontSize = sizeof(kBuiltinFontData);\n"
    "}\n")
configure_file(${AIGE_FONT_CPP}.tmp ${AIGE_FONT_CPP} COPYONLY) # only touched when it changes

add_library(aige_renderer2d STATIC
    Components.h
    Camera2D.h
    ImageData.h
    Renderer2D.h
    Renderer2D.cpp
    SpriteCulling.h
    SpriteCulling.cpp
    Text.h
    Text.cpp
    ${AIGE_FONT_CPP}
)
target_include_directories(aige_renderer2d PUBLIC ${CMAKE_SOURCE_DIR}/engine)
target_include_directories(aige_renderer2d PRIVATE ${stb_SOURCE_DIR})
target_link_libraries(aige_renderer2d PUBLIC aige_rhi aige_ecs)
target_compile_features(aige_renderer2d PUBLIC cxx_std_20)
